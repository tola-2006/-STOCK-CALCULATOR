<script>
  (function() {
    const price = document.getElementById('price');
    const qty = document.getElementById('qty');
    const broker = document.getElementById('broker');
    const chkSEC = document.getElementById('chkSEC');
    const chkNGX = document.getElementById('chkNGX');
    const chkVAT = document.getElementById('chkVAT');
    const chkCSCS = document.getElementById('chkCSCS');
    const chkStamp = document.getElementById('chkStamp');
    const calcBtn = document.getElementById('calcBtn');
    const resultsDiv = document.getElementById('results');
    const totalCost = document.getElementById('totalCost');
    const breakeven = document.getElementById('breakeven');
    const requiredGain = document.getElementById('requiredGain');
    const feeBreakdown = document.getElementById('feeBreakdown');
    const sellPrice = document.getElementById('sellPrice');
    const sellBtn = document.getElementById('sellBtn');
    const sellResult = document.getElementById('sellResult');
    const sellText = document.getElementById('sellText');
    function naira(v) {
      return '₦' + Number(v).toLocaleString('en', {minimumFractionDigits:2, maximumFractionDigits:2});
    }
    function calc() {
      const P = parseFloat(price.value) || 0;
      const Q = parseInt(qty.value) || 0;
      const bRate = parseFloat(broker.value) || 0;
      if (P <= 0 || Q <= 0) { resultsDiv.style.display = 'none'; return; }
      const consideration = P * Q;
      let commission = consideration * (bRate / 100);
      let vat = chkVAT.checked ? commission * 0.075 : 0;
      let sec = chkSEC.checked ? consideration * 0.003 : 0;
      let ngx = chkNGX.checked ? consideration * 0.001674 : 0;
      let cscs = chkCSCS.checked ? 100 : 0;
      let stamp = chkStamp.checked ? 50 : 0;
      let totalFees = commission + vat + sec + ngx + cscs + stamp;
      let totalCostVal = consideration + totalFees;
      let breakEven = totalCostVal / Q;
      let gainPercent = ((breakEven - P) / P) * 100;
      // Show results
      resultsDiv.style.display = 'block';
      totalCost.textContent = naira(totalCostVal);
      breakeven.textContent = naira(breakEven);
      requiredGain.textContent = gainPercent.toFixed(2) + '%';
      let parts = [];
      if (commission > 0) parts.push('Comm ' + naira(commission));
      if (vat > 0) parts.push('VAT ' + naira(vat));
      if (sec > 0) parts.push('SEC ' + naira(sec));
      if (ngx > 0) parts.push('NGX ' + naira(ngx));
      if (cscs > 0) parts.push('CSCS ' + naira(cscs));
      if (stamp > 0) parts.push('Stamp ' + naira(stamp));
      feeBreakdown.textContent = parts.join(' · ') || 'No fees selected';
      // Auto-run sell simulation if sell price is set
      simulateSell(P, Q, breakEven, totalCostVal, commission, vat, sec, ngx, cscs, stamp);
      return { P, Q, breakEven, totalCostVal, totalFees, commission, vat, sec, ngx, cscs, stamp };
    }
    function simulateSell(P, Q, breakEven, totalBuyCost, comm, vat, sec, ngx, cscs, stamp) {
      const sellP = parseFloat(sellPrice.value) || 0;
      if (sellP <= 0 || Q <= 0) { sellResult.style.display = 'none'; return; }
      const sellConsideration = sellP * Q;
      // Sell-side fees (same structure, but broker may differ – we reuse same rates for simplicity)
      let sComm = sellConsideration * (parseFloat(broker.value) || 0) / 100;
      let sVat = chkVAT.checked ? sComm * 0.075 : 0;
      let sSec = chkSEC.checked ? sellConsideration * 0.003 : 0;
      let sNgx = chkNGX.checked ? sellConsideration * 0.001674 : 0;
      let sCscs = chkCSCS.checked ? 100 : 0;
      let sStamp = chkStamp.checked ? 50 : 0;
      let totalSellFees = sComm + sVat + sSec + sNgx + sCscs + sStamp;
      let netProceeds = sellConsideration - totalSellFees;
      let netProfit = netProceeds - totalBuyCost;
      sellResult.style.display = 'block';
      let color = netProfit >= 0 ? '#0b7e3d' : '#c72a2a';
      let sign = netProfit >= 0 ? '+' : '';
      sellText.innerHTML = `
        <strong style="color:${color};">${sign}${naira(netProfit)}</strong> 
        <span style="color:#5e6f8d;font-size:14px;">(sell proceeds ${naira(netProceeds)} – fees ${naira(totalSellFees)})</span>
      `;
    }
    calcBtn.addEventListener('click', function() {
      calc();
    });
    sellBtn.addEventListener('click', function() {
      // Recalc to get latest values, then simulate
      const data = calc();
      if (data) {
        simulateSell(data.P, data.Q, data.breakEven, data.totalCostVal, data.commission, data.vat, data.sec, data.ngx, data.cscs, data.stamp);
      }
    });
    // Auto-calc on any input change (debounced)
    let timer;
    ['input','change'].forEach(ev => {
      [price, qty, broker, sellPrice, chkSEC, chkNGX, chkVAT, chkCSCS, chkStamp].forEach(el => {
        el.addEventListener(ev, function() {
          clearTimeout(timer);
          timer = setTimeout(() => calc(), 300);
        });
      });
    });
    // initial run
    setTimeout(calc, 100);
  })();
</script>
